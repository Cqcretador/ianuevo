import gradio as gr
import PyPDF2
from sentence_transformers import SentenceTransformer
import chromadb
import tempfile

# Inicializa el modelo de embeddings gratuito
embedder = SentenceTransformer("all-MiniLM-L6-v2")

# Inicializa ChromaDB local
chroma_client = chromadb.Client()
collection = chroma_client.get_or_create_collection(name="documentos")

# Función para extraer texto de un PDF
def extract_text_from_pdf(file):
    reader = PyPDF2.PdfReader(file)
    text = ""
    for page in reader.pages:
        text += page.extract_text()
    return text

# Función para hacer preguntas
def responder_pregunta(query):
    # Convertir la pregunta a embedding
    query_embedding = embedder.encode([query]).tolist()[0]
    
    # Buscar documentos relevantes
    results = collection.query(query_embeddings=[query_embedding], n_results=3)
    context_docs = results['documents'][0]
    
    # Armar el contexto (solo los documentos más relevantes)
    context = "\n".join(context_docs)
    return f"Basado en el documento, encontré esta información:\n\n{context}"

# Interfaz de Gradio
iface = gr.Interface(
    fn=responder_pregunta,
    inputs=[gr.Textbox(label="Escribe tu pregunta aquí"), gr.File(label="Sube tu archivo PDF")],
    outputs="text",
    title="Asistente IA con Gradio",
    description="Cargar un archivo PDF y hacerle preguntas"
)

# Lanzar la interfaz
iface.launch()
